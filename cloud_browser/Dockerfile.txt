FROM python:3.13-slim
ARG CHROME_VERSION=157.0.8103.50-1
RUN apt-get update && apt-get install -y --no-install-recommends ca-certificates curl gnupg \
    && curl -fsSL https://dl.google.com/linux/linux_signing_key.pub -o /tmp/google.asc \
    && gpg --show-keys --with-colons /tmp/google.asc | grep -q EB4C1BFD4F042F6DDDCCEC917721F63BD38B4796 \
    && gpg --dearmor -o /usr/share/keyrings/google-linux.gpg /tmp/google.asc \
    && echo 'deb [arch=amd64 signed-by=/usr/share/keyrings/google-linux.gpg] https://dl.google.com/linux/chrome/deb/ stable main' > /etc/apt/sources.list.d/google-chrome.list \
    && apt-get update && apt-get install -y --no-install-recommends google-chrome-stable=${CHROME_VERSION} xvfb openbox x11vnc xauth x11-utils xdotool xclip imagemagick fonts-liberation dbus-x11 \
    && rm -rf /var/lib/apt/lists/* /tmp/google.asc
RUN pip install --no-cache-dir uv==0.13.7
WORKDIR /worker
COPY cloud_browser/pyproject.toml cloud_browser/uv.lock ./
RUN uv sync --frozen --no-dev --no-install-project
COPY cloud_browser/worker ./worker
COPY cloud_browser/THIRD_PARTY_NOTICES.md ./
COPY cloud_browser/licenses ./licenses
RUN useradd --uid 10001 --create-home browser && mkdir -p /data/browser && chown browser:browser /data/browser
ENV PATH="/worker/.venv/bin:$PATH" PYTHONUNBUFFERED=1 PYTHONDONTWRITEBYTECODE=1
EXPOSE 8090
CMD ["sh","-c","mkdir -p /data/browser && chown browser:browser /data/browser && chmod 700 /data/browser && exec setpriv --reuid=10001 --regid=10001 --init-groups uvicorn worker.service:app --host :: --port 8090 --no-access-log --ws-max-size 4194304"]
